if(ANDROID)
    add_library(ote_viewer SHARED main.cpp)
else()
    add_executable(ote_viewer WIN32 MACOSX_BUNDLE main.cpp)
endif()

target_link_libraries(ote_viewer PRIVATE opentakengine)

if(APPLE)
    set_source_files_properties(main.cpp PROPERTIES LANGUAGE OBJCXX)
    target_compile_options(ote_viewer PRIVATE -fobjc-arc)
    if(IOS)
        target_link_libraries(ote_viewer PRIVATE "-framework UIKit" "-framework MetalKit")
    else()
        target_link_libraries(ote_viewer PRIVATE "-framework Cocoa" "-framework MetalKit")
    endif()
elseif(WIN32)
    if(MSVC)
        target_compile_options(ote_viewer PRIVATE /utf-8)
        target_link_options(ote_viewer PRIVATE /ENTRY:mainCRTStartup)
    endif()
elseif(NOT ANDROID)
    find_package(X11 REQUIRED)
    target_link_libraries(ote_viewer PRIVATE X11::X11 X11::Xi X11::Xcursor)
endif()
