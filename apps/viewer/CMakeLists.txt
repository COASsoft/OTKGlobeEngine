if(ANDROID)
    add_library(otk_viewer SHARED main.cpp shot.cpp watchdog.cpp)
else()
    add_executable(otk_viewer WIN32 MACOSX_BUNDLE main.cpp shot.cpp watchdog.cpp win_pointer.cpp)
endif()

target_link_libraries(otk_viewer PRIVATE otkglobeengine)
target_include_directories(otk_viewer PRIVATE ${stb_SOURCE_DIR})

if(APPLE)
    set_source_files_properties(main.cpp PROPERTIES LANGUAGE OBJCXX)
    target_compile_options(otk_viewer PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
    if(IOS)
        target_link_libraries(otk_viewer PRIVATE "-framework UIKit" "-framework MetalKit" "-framework CoreGraphics")
    else()
        target_link_libraries(otk_viewer PRIVATE "-framework Cocoa" "-framework MetalKit" "-framework CoreGraphics")
    endif()
elseif(WIN32)
    if(MSVC)
        target_compile_options(otk_viewer PRIVATE /utf-8)
        target_link_libraries(otk_viewer PRIVATE d3d11 dbghelp opengl32)
    endif()
elseif(NOT ANDROID)
    find_package(X11 REQUIRED)
    target_link_libraries(otk_viewer PRIVATE X11::X11 X11::Xi X11::Xcursor)
endif()
